function icon(name, className) {
    var keys = Object.keys(__icons);
    for (var i = 0; i < keys.length; i++) {
      if (keys[i].toLowerCase() === name.toLowerCase().replace(/-/g, '')) {
        var ic = __icons[keys[i]];
        return '<svg class="icon ' + (className || '') + '" viewBox="' + ic.viewBox + '" width="1em" height="1em" fill="currentColor"><path d="' + ic.path + '"/></svg>';
      }
    }
    return '<span class="icon-missing" style="color:red">[?]' + name + '</span>';
  }
  if (typeof module !== 'undefined' && module.exports) {
    module.exports = { icon: icon };
  }
  if (typeof window !== 'undefined') {
    window.__icons = __icons;
    window.icon = icon;
  }
})();
